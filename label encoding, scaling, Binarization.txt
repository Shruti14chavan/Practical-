import pandas as pd
import numpy as np
# Create a synthetic dataset
np.random.seed(0)
data = pd.DataFrame({
    'Feature1': np.random.choice(['A', 'B', 'C'], 100),
    'Feature2': np.random.randn(100),
    'Feature3': np.random.randint(1, 100, 100),
    'Target': np.random.choice([0, 1], 100)
})
from sklearn.preprocessing import LabelEncoder
# Label encoding for Feature1
label_encoder = LabelEncoder()
data['Feature1_encoded'] = label_encoder.fit_transform(data['Feature1'])
from sklearn.preprocessing import StandardScaler
# Standard scaling for Feature2 and Feature3
scaler = StandardScaler()
data[['Feature2_scaled', 'Feature3_scaled']] = scaler.fit_transform(data[['Feature2',
'Feature3']])
from sklearn.preprocessing import Binarizer
# Binarization for Feature3
binarizer = Binarizer(threshold=50)
data['Feature3_binarized'] = binarizer.fit_transform(data[['Feature3']])
print(data.head())
