# Handle outliers (example: capping values)
# Ensure 'Phone 1' and 'Phone 2' columns contain numeric data
df['Phone 1'] = pd.to_numeric(df['Phone 1'], errors='coerce') # Convert to numeric, invalid values become NaN
df['Phone 2'] = pd.to_numeric(df['Phone 2'], errors='coerce') # Convert to numeric, invalid values become NaN
# Now calculate quantiles and apply clipping 
q1 = df['Phone 1'].quantile(0.25)
q3 = df['Phone 2'].quantile(0.75) 
iqr = q3 - q1
lower_bound = q1 - 1.5 * iqr 
upper_bound = q3 + 1.5 * iqr
df['Phone 1'] = df['Phone 2'].clip(lower_bound, upper_bound)
